# iMessage Integration + CoAgent Category Implementation Plan

> **For Claude:** REQUIRED SUB-SKILL: Use superpowers:executing-plans to implement this plan task-by-task.

**Goal:** Add iMessage as a first-party CoAgent integration with read/send tools, gated behind Full Disk Access, under a new "CoAgent" category in the integrations modal.

**Architecture:** New `packages/mcp-imessage` MCP server reads `~/Library/Messages/chat.db` via SQLite (read-only) and sends via AppleScript/osascript. Appears in integrations modal under "CoAgent" category with `coagent:` slug prefix. Server routes `coagent:` slugs to built-in MCP connect/disconnect flow with FDA permission check.

**Tech Stack:** better-sqlite3 (sync SQLite), @modelcontextprotocol/sdk, AppleScript via child_process.execSync, TypeScript

  { capabilities: { tools: {} } }

server.setRequestHandler(ListToolsRequestSchema, async () => ({
  tools: []

### Task 2: Implement chat.db reader tools

**Files:**
- Modify: `packages/mcp-imessage/src/index.ts`

The chat.db schema has these key tables:
- `message` — ROWID, text, date (CoreData timestamp: nanoseconds since 2001-01-01), is_from_me, handle_id, cache_roomnames
- `handle` — ROWID, id (phone/email), service
- `chat` — ROWID, chat_identifier, display_name
- `chat_message_join` — chat_id, message_id
- `chat_handle_join` — chat_id, handle_id

CoreData timestamps: `(date / 1000000000) + 978307200` converts to Unix epoch seconds.

**Step 1: Add the three read tools to ListToolsRequestSchema**

Replace the empty tools array with:

import Database from 'better-sqlite3'
import { homedir } from 'os'
import { join } from 'path'
import { execSync } from 'child_process'

const CHAT_DB = join(homedir(), 'Library', 'Messages', 'chat.db')

// CoreData timestamp → ISO string
function coreDataToISO(ts: number | null): string | null {
  if (!ts) return null
  return new Date((ts / 1000000000 + 978307200) * 1000).toISOString()
}

function openDb(): Database.Database {
  return new Database(CHAT_DB, { readonly: true, fileMustExist: true })
}

**Step 2: Implement CallToolRequestSchema handler**

```typescript
server.setRequestHandler(CallToolRequestSchema, async (request) => {
  const { name, arguments: args } = request.params

  try {
    if (name === 'list_recent_conversations') {
      const limit = (args?.limit as number) || 20
      const db = openDb()
      const rows = db.prepare(`
        SELECT
          c.chat_identifier,
          c.display_name,
          m.text,
          m.date,
          m.is_from_me,
          h.id AS handle_id
        FROM chat c
        JOIN chat_message_join cmj ON cmj.chat_id = c.ROWID
        JOIN message m ON m.ROWID = cmj.message_id
        LEFT JOIN handle h ON h.ROWID = m.handle_id
        WHERE m.ROWID = (
          SELECT MAX(m2.ROWID) FROM message m2
          JOIN chat_message_join cmj2 ON cmj2.message_id = m2.ROWID
          WHERE cmj2.chat_id = c.ROWID
        )
        ORDER BY m.date DESC
        LIMIT ?
      `).all(limit) as any[]
      db.close()

      const conversations = rows.map(r => ({
        identifier: r.chat_identifier,
        displayName: r.display_name || r.handle_id || r.chat_identifier,
        lastMessage: r.text?.slice(0, 100) || '(attachment)',
        date: coreDataToISO(r.date),
        fromMe: !!r.is_from_me
      }))
      return { content: [{ type: 'text', text: JSON.stringify(conversations, null, 2) }] }
    }

    if (name === 'get_conversation') {
      const identifier = args?.identifier as string
      const limit = (args?.limit as number) || 50
      const db = openDb()
      const rows = db.prepare(`
        SELECT m.text, m.date, m.is_from_me, h.id AS sender
        FROM message m
        JOIN chat_message_join cmj ON cmj.message_id = m.ROWID
        JOIN chat c ON c.ROWID = cmj.chat_id
        LEFT JOIN handle h ON h.ROWID = m.handle_id
        WHERE c.chat_identifier LIKE ?
        ORDER BY m.date DESC
        LIMIT ?
      `).all(`%${identifier}%`, limit) as any[]
      db.close()

      const messages = rows.reverse().map(r => ({
        text: r.text || '(attachment)',
        date: coreDataToISO(r.date),
        fromMe: !!r.is_from_me,
        sender: r.is_from_me ? 'me' : (r.sender || 'unknown')
      }))
      return { content: [{ type: 'text', text: JSON.stringify(messages, null, 2) }] }
    }

    if (name === 'search_messages') {
      const query = args?.query as string
      const contact = args?.contact as string | undefined
      const limit = (args?.limit as number) || 30
      const db = openDb()

      let sql = `
        SELECT m.text, m.date, m.is_from_me, h.id AS sender, c.chat_identifier
        FROM message m
        JOIN chat_message_join cmj ON cmj.message_id = m.ROWID
        JOIN chat c ON c.ROWID = cmj.chat_id
        LEFT JOIN handle h ON h.ROWID = m.handle_id
        WHERE m.text LIKE ?
      `
      const params: any[] = [`%${query}%`]

      if (contact) {
        sql += ` AND c.chat_identifier LIKE ?`
        params.push(`%${contact}%`)
      }
      sql += ` ORDER BY m.date DESC LIMIT ?`
      params.push(limit)

      const rows = db.prepare(sql).all(...params) as any[]
      db.close()

      const results = rows.map(r => ({
        text: r.text,
        date: coreDataToISO(r.date),
        fromMe: !!r.is_from_me,
        sender: r.is_from_me ? 'me' : (r.sender || 'unknown'),
        conversation: r.chat_identifier
      }))
      return { content: [{ type: 'text', text: JSON.stringify(results, null, 2) }] }
    }

    if (name === 'send_message') {
      const to = args?.to as string
      const text = args?.text as string

      // Sanitize for AppleScript — escape backslashes and quotes
      const escaped = text.replace(/\\/g, '\\\\').replace(/"/g, '\\"')
      const script = `tell application "Messages"
  set targetService to 1st account whose service type = iMessage
  set targetBuddy to participant "${to}" of targetService
  send "${escaped}" to targetBuddy
end tell`

      try {
        execSync(`osascript -e '${script.replace(/'/g, "'\\''")}'`, {
          timeout: 10000,
          stdio: 'pipe'
        })
        return { content: [{ type: 'text', text: `Message sent to ${to}` }] }
      } catch (err: any) {
        return { content: [{ type: 'text', text: `Failed to send: ${err.stderr?.toString() || err.message}` }], isError: true }
      }
    }

    throw new Error(`Unknown tool: ${name}`)
  } catch (err: any) {
    // Surface FDA errors clearly
    if (err.code === 'SQLITE_CANTOPEN') {
      return {
        content: [{ type: 'text', text: 'Cannot access iMessage database. Full Disk Access may have been revoked. Ask the user to re-enable it in System Settings > Privacy & Security > Full Disk Access.' }],
        isError: true
      }
    }
    return { content: [{ type: 'text', text: `Error: ${err.message}` }], isError: true }
  }
})
```
